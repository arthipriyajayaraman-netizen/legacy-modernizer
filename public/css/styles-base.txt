  function toggleRow(id){
    const chk = document.getElementById('chk-'+id);
    const row = document.getElementById('row-'+id);
    if(selected.has(id)){
      selected.delete(id);
      chk.classList.remove('checked');
      row.classList.remove('selected');
    } else {
      selected.add(id);
      chk.classList.add('checked');
      row.classList.add('selected');
    }
    document.getElementById('selCount').textContent = selected.size;
    document.getElementById('modernizeBtn').disabled = selected.size === 0;
  }

  function startScan(){
    const status = document.getElementById('scanStatus');
    status.style.display = 'flex';
    document.getElementById('scanStatusText').textContent = 'Cloning repository and analyzing dependency manifests…';
    setTimeout(() => {
      goToStep(2);
    }, 900);
  }

  function applyModernization(){
    const panel = document.getElementById('changesPanel');
    const list = document.getElementById('fileDiffList');
    let rows = [];
    selected.forEach(id => {
      (fileChanges[id] || []).forEach(fc => rows.push(fc));
    });
    list.innerHTML = rows.map(fc => `
      <div class="file-diff">
        <span class="fname">${fc.f}</span>
        <span class="diff-stats"><span class="add">+${fc.add}</span><span class="rem">−${fc.rem}</span></span>
      </div>
    `).join('');
    panel.classList.add('show');

    // update publish-step summary
    const totalAdd = rows.reduce((s,r)=>s+r.add,0);
    const totalRem = rows.reduce((s,r)=>s+r.rem,0);
    document.getElementById('sumComponents').textContent = selected.size;
    document.getElementById('sumFiles').textContent = rows.length;
    document.getElementById('sumAdd').textContent = '+'+totalAdd;
    document.getElementById('sumRem').textContent = '−'+totalRem;

    // keep the publish-step commit message and PR title consistent with what was actually selected
    const suggestedNames = assetData.filter(a => selected.has(a.id)).map(a => a.suggested);
    const shortList = suggestedNames.slice(0,3).join(', ') + (suggestedNames.length > 3 ? ', …' : '');

      `Modernize: ${shortList}\n\nApplied via Legacy Modernizer agent — ${selected.size} component${selected.size===1?'':'s'} updated.`;
    document.querySelector('.pr-link').textContent = `🔗 #482 — Modernize: ${shortList}`;

    document.getElementById('modernizeBtn').disabled = true;
  }

      ? 'A pull request was opened against main.'

    document.querySelector('.pr-link').style.display = prOn ? 'inline-block' : 'none';
    state.classList.add('show');
  }

    renderRows();
    document.getElementById('selCount').textContent = '0';
    document.getElementById('modernizeBtn').disabled = true;
    document.getElementById('changesPanel').classList.remove('show');
    document.getElementById('publishForm').classList.remove('hidden');
    document.getElementById('successState').classList.remove('show');
    goToStep(1);
  }

  function goToStep(n){
    for(let i=1;i<=3;i++){
      document.getElementById('screen-'+i).classList.toggle('active', i===n);
      const nav = document.getElementById('stepnav-'+i);
      nav.classList.toggle('active', i===n);
      nav.classList.toggle('done', i<n);
    }
    window.scrollTo({top:0, behavior:'smooth'});
  }
